         POPUP JAVASCRIPT â€” Fixed class-based show/hide
    ============================================================ -->
    <script>
    (function () {
        var POPUPS = ['barsana', 'hathras'];
        var idx = 0;
        var closing = false;

        function show(id) {
            var el = document.getElementById('nlp-' + id);
            if (!el) return;
            // Remove old animation so it replays
            var card = el.querySelector('.nlp-card');
            if (card) { card.style.animation = 'none'; card.offsetHeight; card.style.animation = ''; }
            el.classList.add('nlp-show');
            document.body.style.overflow = 'hidden';
            closing = false;
        }

        function close(id) {
            if (closing) return;
            closing = true;
            var el = document.getElementById('nlp-' + id);
            if (!el) return;
            var card = el.querySelector('.nlp-card');
            if (card) {
                card.style.transition = 'opacity .22s ease, transform .22s ease';
                card.style.opacity = '0';
                card.style.transform = 'translateY(18px) scale(.96)';
            }
            setTimeout(function () {
                el.classList.remove('nlp-show');
                if (card) { card.style.opacity = ''; card.style.transform = ''; card.style.transition = ''; }
                idx++;
                if (idx < POPUPS.length) {
                    show(POPUPS[idx]);
                } else {
                    document.body.style.overflow = '';
                }
            }, 240);
        }

        // Expose globally
        window.nlpClose = close;

        // Wire up close buttons and backdrops
        function wire() {
            POPUPS.forEach(function (id) {
                var xBtn = document.getElementById('nlp-x-' + id);
                var bd   = document.getElementById('nlp-bd-' + id);
                if (xBtn) xBtn.addEventListener('click', function (e) { e.stopPropagation(); close(id); });
                if (bd)   bd.addEventListener('click', function () { close(id); });
            });
            // ESC key
            document.addEventListener('keydown', function (e) {
                if (e.key === 'Escape' && idx < POPUPS.length) close(POPUPS[idx]);
            });
        }

        // Show first popup 1.4s after page load
        if (document.readyState === 'complete') {
            wire();
            setTimeout(function () { show(POPUPS[0]); }, 1400);
        } else {
            window.addEventListener('load', function () {
                wire();
                setTimeout(function () { show(POPUPS[0]); }, 1400);
            });
        }
    })();
    </script>
